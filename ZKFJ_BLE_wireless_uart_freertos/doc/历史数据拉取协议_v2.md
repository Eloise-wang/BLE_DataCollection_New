# 历史数据拉取协议 v2.0

> 适用于：下位机固件 + 上位机应用层通信
>
> 修订日期：2026-06-01
>
> **本协议直接增强 0x04 命令，替代原有连续发送模式。**

---

## 1. 概述

本协议用于上位机从下位机 NOR Flash 中拉取历史采集数据。设计目标：

- **可靠**：支持丢帧检测与精确重传
- **快速**：突发模式减少请求往返次数
- **兼容**：数据帧格式与原有 0x04 协议完全兼容，上位机解析逻辑无需修改

---

## 2. 通信模式

**请求-响应模式（带突发窗口）**

```
上位机 ──── GET[offset, count] ─────→ 下位机
          ↑                          ↓
          ←─── DATA[offset, data] ←──  （连续发 count 帧后停止）
          ↑                          ↓
上位机收到 count 帧（或检测到丢帧）───→ 发起下一次请求
```

### 关键特性

- 下位机收到一次 GET 请求后，连续发送 `count` 帧后**停止**，等待下一个请求
- 每帧携带 `offset`（本帧数据起始字节位置），上位机按 offset 拼接
- 丢帧通过 offset 间隙检测，上位机从断点发起重传请求
- BLE 层保留 3ms 帧间隔，防止控制器 TX 缓冲区满导致丢帧

## 3. 帧格式

### 3.1 上位机 → 下位机：请求帧（GET）

```
┌────────┬──────────────┬───────────────────────┬─────────────────┐
│ CMD    │ task_id      │ offset                 │ count           │
│ 1字节   │ 8字节LE     │ 4字节LE               │ 2字节LE         │
└────────┴──────────────┴───────────────────────┴─────────────────┘
                                                      ↑ 
                                             file_sel 隐含在 count 高位：
                                             count & 0x00FF = 请求帧数（1~255）
                                             count >> 8    = file_sel (0~3)
```

**注意**：本协议复用 0x04 命令码，请求帧格式从 15 字节变为 14 字节：
- 原 0x04 格式：`[task_id(8)][offset(4)][max_bytes(2)][file_sel(1)]` = 15 字节
- 新 0x04 格式：`[task_id(8)][offset(4)][count(2)]` = 14 字节

下位机根据 payload 长度区分新旧协议：
- `len == 14`：新协议（带 count）
- `len == 15`：旧协议（忽略 max_bytes，连续发送）

```
CMD      = 0x04（历史数据请求，增强版）
task_id  = 任务ID（64位）
offset   = 请求从第几个字节开始（0 = 从头开始）
count    = 请求多少帧（1~255，建议 50）
```

**帧示例（hex）**：

```
新协议请求 Data 区，从 0 字节开始，请求 50 帧：
04 00 00 00 00 00 00 00 00 00 00 00 00 32 00

04            = CMD = 0x04
00 00 00 00   = task_id = 0
00 00 00 00   = offset = 0（从头开始）
32 00         = count = 50（0x0032 = 50）
（无 file_sel 字段，默认为 0 = Data 区）
```

### 3.2 下位机 → 上位机：数据帧（DATA）

**帧格式与旧协议完全一致**，无需修改上位机解析逻辑。

```
┌────────┬──────────┬──────────────┬──────────────────────┬───────────────────────────┬────────┐
│ SOF    │ len      │ task_id       │ offset                │ total_bytes               │ data   │ CRC16  │
│ 1字节   │ 2字节LE │ 8字节LE      │ 4字节LE               │ 4字节LE                   │ N字节   │ 2字节LE│
└────────┴──────────┴──────────────┴──────────────────────┴───────────────────────────┴────────┴────────┘

SOF         = 0x5A（帧起始标记）
len         = payload 长度 = 16 + N（不含 SOF 和 len 本身）
task_id     = 任务ID（64位）
offset      = 本帧数据在文件中的起始字节位置
total_bytes = 文件总字节数（固定值，同一 task_id 下所有帧相同）
data        = 业务数据（每帧最多 244 字节，BLE MTU 上限）
CRC16       = 从 len 第一个字节到 data 最后一个字节的 CRC-16/MODBUS 校验
```

**帧示例（hex）**：

```
5A 58 01                              ← SOF + len = 0x0158 = 344（payload长度）
00 00 00 00 00 00 00 00              ← task_id = 0x0000000000000000
00 00 00 00                           ← offset = 0（从头开始）
00 C0 0B 00                           ← total_bytes = 0x000BC000 = 770048
[244 字节 data]                       ← 业务数据
XX XX                                 ← CRC16
```

### 3.3 下位机 → 上位机：ACK 响应

下位机收到 GET 请求后，先发送 ACK：

```
┌────────┬──────────┐
│ SOF    │ status    │
│ 1字节   │ 1字节     │
└────────┴──────────┘

SOF    = 0xA5（ACK 帧标记）
status = 0x00 = OK
       = 0x01 = 任务未找到
       = 0x02 = 参数错误
       = 0x03 = 存储错误
```

## 4. 上位机实现指南

### 4.1 协议区分规则

```
请求帧长度 = 14 字节 → 新协议（带 count，突发窗口）
请求帧长度 = 15 字节 → 旧协议（忽略 max_bytes，连续发送）

新协议请求帧格式：
  [CMD(1)][task_id(8)][offset(4)][count(2)]

旧协议请求帧格式：
  [CMD(1)][task_id(8)][offset(4)][max_bytes(2)][file_sel(1)]  ← 共 16 字节（含 CMD）
```

### 4.2 状态机

```python
class HistoryPuller:
    def __init__(self, max_retry=3, burst_count=50):
        self.offset = 0               # 当前已确认收到的字节偏移
        self.burst_count = burst_count
        self.max_retry = max_retry

    def pull_all(self, task_id):
        # 发新协议请求（14字节，不含 file_sel，默认为 Data 区）
        self.send_get_v2(task_id, self.offset, self.burst_count)
        received = []

        while True:
            frame = self.recv_frame()  # 阻塞接收

            if frame is None:
                break  # 连接断开

            # CRC 校验
            if not self.check_crc(frame):
                continue  # CRC 错误，等待下一帧

            frame_offset = frame["offset"]

            # 收到帧在突发窗口内的判断
            if frame_offset < self.offset:
                continue  # 重复帧，忽略

            if frame_offset == self.offset:
                # 正常帧：存储并推进 offset
                received.append(frame["data"])
                self.offset += len(frame["data"])

            elif frame_offset > self.offset:
                # 丢帧检测：offset 有跳跃
                # 例：期望 offset=9600，收到 offset=9800 → 丢失 offset=9600~9799 的数据
                lost_bytes = frame_offset - self.offset
                print(f"[WARN] 检测到丢帧，丢失 {lost_bytes} 字节，从 offset={frame_offset} 重传")
                self.offset = frame_offset
                self.send_get_v2(task_id, self.offset, self.burst_count)
                received.append(frame["data"])
                self.offset += len(frame["data"])

            # 突发窗口收满，发下一个请求
            if len(received) >= self.burst_count:
                self.send_get_v2(task_id, self.offset, self.burst_count)
                received = []

            # 传输结束判断
            if self.offset >= frame["total_bytes"]:
                break

    def send_get_v2(self, task_id, offset, count):
        """新协议：14字节请求帧"""
        payload = struct.pack("<QIH",
            task_id,   # Q = uint64_t
            offset,    # I = uint32_t
            count,     # H = uint16_t（1~255）
        )
        self.send_bytes(bytes([0x04]) + payload)

    def send_get_v1(self, task_id, file_sel=0):
        """旧协议：15字节请求帧（兼容）"""
        payload = struct.pack("<QIB",
            task_id,
            0,          # offset = 0
            file_sel,   # file_sel = 0 = Data 区
        )
        self.send_bytes(bytes([0x04]) + payload)

    def check_crc(self, frame):
        # CRC16/MODBUS: 从 len 第一个字节到 data 最后一个字节
        crc_range = frame["raw"][1:-2]
        return self.calc_crc16(crc_range) == frame["crc"]

    def calc_crc16(self, data: bytes) -> int:
        crc = 0xFFFF
        for b in data:
            crc ^= b
            for _ in range(8):
                crc = (crc >> 1) ^ 0xA001 if (crc & 1) else crc >> 1
        return crc
```

### 4.3 重传逻辑

```
初始请求: GET[offset=0, count=50]

收到 48 帧后，下一帧期望 offset=9600 但收到 offset=9800 → 检测丢帧
→ 发 GET[offset=9600, count=2]（从丢帧位置重传 2 帧）
→ 收到 offset=9600 的帧 ✅
→ 收到 offset=9800 的帧 ✅
→ 继续发 GET[offset=10000, count=50]（正常继续）
```

### 4.4 突发窗口大小建议

| 场景 | 建议 count | 说明 |
|------|-----------|------|
| 稳定环境（实验室） | 100 | 丢帧率低，效率优先 |
| 一般环境（办公室） | 50 | 平衡效率与可靠性 |
| 恶劣环境（工厂） | 20 | 丢帧率高，频繁续传，小窗口减少浪费 |

## 5. 下位机实现指南

### 5.1 协议区分逻辑

```c
static void proto_handle_request_history(const proto_cmd_msg_t *msg)
{
    /* 根据 payload 长度区分新旧协议 */
    if (msg->len == 14U)
    {
        /* 新协议：带 count，突发窗口模式 */
        proto_handle_request_history_v2(msg);
    }
    else if (msg->len == 15U)
    {
        /* 旧协议：忽略 max_bytes，连续发送 */
        proto_handle_request_history_legacy(msg);
    }
    else
    {
        proto_send_ack(msg->cmd, PROTO_STATUS_PARAM_ERROR);
    }
}
```

### 5.2 关键代码修改点

**① proto_cmd_handler.c：新增宏定义**

```c
// BLE MTU=247，ATT payload 最大 244 字节
#define PROTO_MAX_CHUNK_SIZE  244U

// 每帧最大业务数据字节数（BLE MTU 上限）
static uint8_t s_data_buf[PROTO_MAX_CHUNK_SIZE];

// DATA 帧最大长度：SOF(1) + len(2) + task_id(8) + offset(4) + total(4) + data(244) + CRC(2)
static uint8_t s_frame_buf[1 + 2 + 8 + 4 + 4 + PROTO_MAX_CHUNK_SIZE + 2];
```

**② proto_cmd_handler.c：修改 proto_handle_request_history**

```c
static void proto_handle_request_history(const proto_cmd_msg_t *msg)
{
    if (msg->len == 14U)
    {
        /* 新协议：突发窗口模式 */
        proto_handle_request_history_v2(msg);
    }
    else if (msg->len == 15U)
    {
        /* 旧协议：连续发送模式（兼容） */
        proto_handle_request_history_legacy(msg);
    }
    else
    {
        proto_send_ack(msg->cmd, PROTO_STATUS_PARAM_ERROR);
    }
}

static void proto_handle_request_history_v2(const proto_cmd_msg_t *msg)
{
    /* 解析请求参数（14字节） */
    uint64_t task_id;
    uint32_t offset;
    uint16_t count;

    (void)proto_payload_read_u64_le(&msg->payload[0], msg->len, &task_id);
    (void)proto_payload_read_u32_le(&msg->payload[8], msg->len - 8U, &offset);

    /* count 在 payload[12]~[13]（uint16_t LE） */
    count = ((uint16_t)msg->payload[12]) | ((uint16_t)msg->payload[13] << 8);
    if (count == 0U || count > 50U) count = 50U;  /* 限制上限 50 帧 */

    /* 文件选择：默认 Data 区（0） */
    const uint8_t file_sel = 0U;

    /* 获取数据总大小 */
    uint32_t total_bytes = 0U;
    if (!APP_Storage_Init() || !APP_Storage_GetDataSize(task_id, &total_bytes))
    {
        proto_send_ack(msg->cmd, PROTO_STATUS_STORAGE_ERROR);
        return;
    }

    if (offset >= total_bytes)
    {
        proto_send_ack(msg->cmd, PROTO_STATUS_PARAM_ERROR);
        return;
    }

    /* 发 ACK */
    proto_send_ack(msg->cmd, PROTO_STATUS_OK);

    /* 读取并逐帧发送，直到发完 count 帧或数据结束 */
    uint32_t pos = offset;
    uint16_t frames_sent = 0U;

    while (frames_sent < count && pos < total_bytes)
    {
        uint16_t to_read = (uint16_t)((total_bytes - pos < PROTO_MAX_CHUNK_SIZE)
                                      ? (total_bytes - pos) : PROTO_MAX_CHUNK_SIZE);

        int nread = APP_Storage_ReadData(task_id, pos, s_data_buf, to_read);
        if (nread <= 0) break;

        size_t out_len = 0U;
        if (!PROTO_DataBuildFrame(task_id, pos, total_bytes,
                                   s_data_buf, (uint16_t)nread,
                                   s_frame_buf, sizeof(s_frame_buf), &out_len))
        {
            break;
        }

        proto_send(s_frame_buf, out_len);

        pos += (uint32_t)nread;
        frames_sent++;
    }
    /* 发完即停止，等待下一个 GET 请求 */
}
```

## 7. 错误处理

### 7.1 上位机错误处理

| 错误 | 处理方式 |
|------|---------|
| CRC 校验失败 | 忽略该帧，等待重传 |
| 帧丢失（offset 跳跃）| 发 GET[offset=丢失位置, count=2] 重传 |
| 连续 3 次丢帧 | 记录日志，尝试重连 |
| 连接断开 | 重新连接后，从上次确认的 offset 继续 |

### 7.2 下位机错误处理

| 错误 | 处理方式 |
|------|---------|
| task_id 不存在 | 发 ACK status=0x01 |
| offset 越界 | 发 ACK status=0x02 |
| Flash 读取失败 | 发 ACK status=0x03 |
| BLE 断开连接 | 停止发送，等待重连 |
